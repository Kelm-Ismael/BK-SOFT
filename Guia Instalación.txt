clinetes y usuarios relacionados sql 
(-- Cada usuario con su cliente correspondiente (deberían coincidir los id)
SELECT
    u.id_usuario,
    u.email,
    u.estado_cuenta,
    c.id_cliente,
    c.nombre,
    c.apellido
FROM usuarios u
JOIN clientes c ON c.id_cliente = u.id_usuario
ORDER BY u.id_usuario;

-- Usuarios que NO tienen fila en clientes (no debería haber ninguno,
-- porque el registro crea ambas filas juntas)
SELECT u.id_usuario, u.email, u.estado_cuenta
FROM usuarios u
LEFT JOIN clientes c ON c.id_cliente = u.id_usuario
WHERE c.id_cliente IS NULL;)

clientes sso_google
(SELECT u.id_usuario, u.email, u.estado_cuenta, c.id_google_sso
FROM usuarios u
JOIN clientes c ON c.id_cliente = u.id_usuario
WHERE c.id_google_sso IS NOT NULL;)

1. BACKEND

cd backend

# Inicializa el proyecto Node (crea package.json)
npm init -y

# --- Dependencias de PRODUCCIÓN (necesarias para que la app corra) ---
npm install express cors pg dotenv

#    express    -> framework HTTP para armar los endpoints (POST /registro, etc.)
#    cors       -> habilita que el frontend (otro origen/puerto) le pegue a esta API
#    pg         -> driver nativo de PostgreSQL para Node.js
#    dotenv     -> carga las variables del archivo .env (DATABASE_URL, etc.)

# --- Dependencias de DESARROLLO (solo se usan mientras programás) ---
npm install -D typescript tsx @types/express @types/cors @types/node @types/pg
npm install bcryptjs
npm install nodemailer && npm install --save-dev @types/nodemailer
npm install --save-dev jest ts-jest @types/jest     --> (framework)


#    typescript     -> compilador/tipado de TS
#    tsx            -> ejecuta .ts directo en dev con recarga automática
#    @types/express -> tipos de TS para Express
#    @types/cors    -> tipos de TS para cors
#    @types/node    -> tipos de TS para las APIs nativas de Node (fs, crypto, etc.)
#    @types/pg      -> tipos de TS para el driver de PostgreSQL
#    npm install --save-dev @types/bcryptjs
#   npm install nodemailer && npm install --save-dev @types/nodemailer


# Genera tsconfig.json (configuración del compilador TS)
npx tsc --init
